DROP DATABASE IF EXISTS estagios;
CREATE DATABASE estagios;
USE estagios;

-- Tabela de Alunos
CREATE TABLE IF NOT EXISTS aluno (
  matricula INT AUTO_INCREMENT PRIMARY KEY,
  endereco_aluno VARCHAR(100),
  nome_aluno VARCHAR(75) NOT NULL,
  idade INT NOT NULL,
  email VARCHAR(80) UNIQUE NOT NULL,
  nome_escola VARCHAR(70) NOT NULL,
  senha VARCHAR(255) NOT NULL
);

-- Tabela de Empresas
CREATE TABLE IF NOT EXISTS empresa (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nome_empresa VARCHAR(70) UNIQUE NOT NULL,
  endereco_empresa VARCHAR(80) NOT NULL,
  email_empresa VARCHAR(120) UNIQUE NOT NULL,
  senha VARCHAR(255) NOT NULL,
  cnpj VARCHAR(18) UNIQUE NOT NULL
);

-- Tabela de Vagas
CREATE TABLE IF NOT EXISTS vagas (
  vagas_id INT AUTO_INCREMENT PRIMARY KEY,
  titulo VARCHAR(100) NOT NULL,
  tipoDeVaga VARCHAR(50) NOT NULL,
  area VARCHAR(90) NOT NULL,
  descricao VARCHAR(1800) NOT NULL,
  salario DECIMAL(15,2) NOT NULL,
  endereco VARCHAR(90) NOT NULL,
  estado VARCHAR(90) NOT NULL,
  atividades VARCHAR(1800) NOT NULL,
  requisitos VARCHAR(1800) NOT NULL,
  horario VARCHAR(90) NOT NULL,
  id_empresa INT NOT NULL,
  matricula_aluno INT,
  CONSTRAINT fk_vagas_empresa FOREIGN KEY (id_empresa) REFERENCES empresa(id) ON DELETE CASCADE,
  CONSTRAINT fk_vagas_aluno FOREIGN KEY (matricula_aluno) REFERENCES aluno(matricula)
);

-- Tabela de Candidaturas (aluno pode se candidatar a várias vagas)
CREATE TABLE IF NOT EXISTS candidaturas (
  id INT AUTO_INCREMENT PRIMARY KEY,
  matricula INT NOT NULL,
  vagas_id INT NOT NULL,
  data_candidatura TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (matricula) REFERENCES aluno(matricula) ON DELETE CASCADE,
  FOREIGN KEY (vagas_id) REFERENCES vagas(vagas_id) ON DELETE CASCADE,
  UNIQUE KEY unica_candidatura (matricula, vagas_id)
);
